  7  |  * then check for idle network with a timeout. This is more reliable for testing

  9  |  */
  10 | export async function waitForPageLoad (page: Page): Promise<void> {
  11 |   try {
  12 |     // First, wait for DOM to be loaded
  13 |     await page.waitForLoadState('domcontentloaded')

  15 |     // Then attempt to wait for network idle with a custom timeout.
  16 |     // If it fails (e.g., due to infinite polling), we continue anyway
  17 |     // as the page is likely usable even if some background requests are pending.

  20 |       const networkIdlePromise = page.waitForLoadState('networkidle')
  21 |       const timeoutPromise = new Promise<never>((_, reject) => {
  22 |         setTimeout(() => reject(new Error('Network idle timeout')), NETWORK_IDLE_TIMEOUT)
  23 |       })
  24 |       await Promise.race([networkIdlePromise, timeoutPromise])
